class Solution {
    public double findMaxAverage(int[] arr, int k) {
        
        double maxAvg;
        int maxSum = Integer.MIN_VALUE;
        int n=arr.length;
        
        //BRUTE - O(n2)
        // for(int i=0;i<=n-k;i++){
        //     int sum=0;
        //     for(int j=i;j<i+k;j++){
        //         sum+=arr[j];
        //     }
        //     if(sum>maxSum) maxSum=sum;
        // }

        //OTIMIZATION - sliding window O(n)
        int sum=0;
        //sum of first window
        for(int i=0;i<k;i++){
                sum+=arr[i];
        }

        maxSum=sum;

        for(int i=k;i<n;i++){
            sum+=arr[i];
            sum-=arr[i-k];
            if(sum>maxSum) maxSum=sum;
        }
        
        maxAvg = (double)maxSum / k;

        return maxAvg;
    }
}
